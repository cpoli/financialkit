"""Synthetic returns from a linear factor model, for tests and examples."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Optional

import numpy as np
from numpy.typing import ArrayLike

__all__ = ["FactorPanel", "simulate_factor_model"]

@dataclass
class FactorPanel:
    """Excess returns generated by :math:`r^e_t = \\alpha + B f_t + \\varepsilon_t`."""

    excess_returns: np.ndarray
    """ndarray, shape (T, n)."""

    factors: np.ndarray
    """ndarray, shape (T, k): Factor excess returns."""

    betas: np.ndarray
    """ndarray, shape (n, k): True loadings."""

    alphas: np.ndarray
    """ndarray, shape (n,): True pricing errors."""

def simulate_factor_model(
    n_periods: int,
    betas: ArrayLike,
    factor_means: ArrayLike,
    factor_cov: ArrayLike,
    idio_vol: ArrayLike = 0.05,
    alphas: Optional[ArrayLike] = None,
    rng: np.random.Generator | int | None = None,
) -> FactorPanel:
    """Normal factor returns and independent normal residuals.

    Parameters
    ----------
    n_periods : int
    betas : array_like, shape (n,) or (n, k)
    factor_means : array_like, shape (k,)
        Expected factor excess returns, the premia when the factors are traded.
    factor_cov : array_like, shape (k, k)
    idio_vol : array_like, shape (n,) or scalar
        Residual volatility per period.
    alphas : array_like, shape (n,), optional
        Pricing errors; zero by default, so the model holds.
    rng : numpy.random.Generator or int or None

    Returns
    -------
    FactorPanel

    Examples
    --------
    >>> panel = simulate_factor_model(120, [0.5, 1.0, 1.5], [0.006], [[0.002]], rng=0)
    >>> panel.excess_returns.shape, panel.factors.shape
    ((120, 3), (120, 1))
    """
    rng = np.random.default_rng(rng)
    B = np.asarray(betas, dtype=np.float64)
    B = B[:, None] if B.ndim == 1 else B
    n, k = B.shape
    mu = np.atleast_1d(np.asarray(factor_means, dtype=np.float64))
    V = np.atleast_2d(np.asarray(factor_cov, dtype=np.float64))
    a = np.zeros(n) if alphas is None else np.broadcast_to(np.asarray(alphas, dtype=np.float64), (n,)).copy()
    f = rng.multivariate_normal(mu, V, size=n_periods, method="cholesky")
    eps = rng.standard_normal((n_periods, n)) * np.broadcast_to(np.asarray(idio_vol, dtype=np.float64), (n,))
    return FactorPanel(excess_returns=a + f @ B.T + eps, factors=f, betas=B, alphas=a)
